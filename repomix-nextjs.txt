import type { MissionEnvironment, VitalReadings } from "@/types/health";

/** Generate hourly (or sub-hour) samples for replay drill-down on a selected mission day. */
export function buildIntradaySamples(
  daily: VitalReadings,
  slots = 24
): VitalReadings[] {
  const anchor = new Date(daily.recordedAt);
  const dayStart = new Date(
    Date.UTC(
      anchor.getUTCFullYear(),
      anchor.getUTCMonth(),
      anchor.getUTCDate(),
      0,
      0,
      0,
      0
    )
  );

  const samples: VitalReadings[] = [];
  const minutesPerSlot = Math.floor((24 * 60) / slots);

  for (let i = 0; i < slots; i += 1) {
    const t = new Date(dayStart);
    t.setUTCMinutes(t.getUTCMinutes() + i * minutesPerSlot);
    t.setUTCSeconds(15 + (i % 45));

    const phase = i / slots;
    const wobble = Math.sin(phase * Math.PI * 2);
    const micro = Math.cos(phase * Math.PI * 4) * 0.5;

    samples.push({
      ...daily,
      recordedAt: t.toISOString(),
      systolicBP: Math.round(daily.systolicBP + wobble * 4 + micro),
      diastolicBP: Math.round(daily.diastolicBP + wobble * 2),
      heartRate: Math.round(daily.heartRate + wobble * 5 + (i % 3)),
      bloodSugar: Math.round(daily.bloodSugar + wobble * 6),
      temperature: Math.round((daily.temperature + micro * 0.15) * 10) / 10,
      sleepHours: Math.max(
        0,
        Math.round((daily.sleepHours * (0.92 + phase * 0.08) + micro) * 10) / 10
      ),
      radiationMsv:
        daily.radiationMsv !== undefined
          ? Math.round((daily.radiationMsv / slots + wobble * 0.02) * 1000) / 1000
          : undefined,
      moodScore: Math.min(
        5,
        Math.max(1, daily.moodScore + (wobble > 0.6 ? 1 : wobble < -0.6 ? -1 : 0))
      ),
      stressLevel:
        daily.stressLevel !== undefined
          ? Math.min(10, Math.max(1, Math.round(daily.stressLevel + wobble * 1.5)))
          : undefined,
      extraMetrics: {
        ...daily.extraMetrics,
        intradaySlot: String(i),
      },
    });
  }

  return samples;
}

export function missionLogEventsForDay(
  missionDay: number,
  environment: MissionEnvironment
): string {
  const codes: string[] = ["routine_vitals", "habitat_air_check"];
  if (missionDay % 3 === 0) codes.push("resistive_exercise_block");
  if (missionDay % 5 === 0) codes.push("hydration_protocol");
  if (missionDay === 12) codes.push("solar_particle_watch");
  if (missionDay === 13) codes.push("circadian_light_therapy");
  if (missionDay === 14) codes.push("ground_sync_window");
  if (environment === "rover" || environment === "lunar_base") {
    codes.push("eva_suit_prep");
  }
  if (environment === "mars_base" || environment === "spacecraft") {
    codes.push("autonomous_med_briefing");
  }
  return codes.join(",");
}

import { buildMockMissionHistory } from "@/data/mockMissionHistory";
import { buildIntradaySamples } from "@/data/intradaySamples";
import type { AppLocale } from "@/lib/locale";
import { formatDayAxis, formatIntradayAxis } from "@/lib/missionTime";
import type { ChartTimeGranularity, MissionEnvironment, VitalReadings } from "@/types/health";

export interface ChartTimelineRow {
  axisLabel: string;
  missionDay: number;
  recordedAt: string;
  systolicBP: number;
  heartRate: number;
  sleepHours: number;
  bloodSugar: number;
  moodScore: number;
  radiationMsv: number;
  radiationCumulative: number;
  reading: VitalReadings;
}

const INTRADAY_SLOTS: Record<Exclude<ChartTimeGranularity, "days">, number> = {
  hours: 24,
  minutes: 60,
};

function axisLabelForGranularity(
  reading: VitalReadings,
  granularity: ChartTimeGranularity,
  locale: AppLocale
): string {
  if (granularity === "days") {
    return formatDayAxis(reading.missionDay, reading.recordedAt, locale);
  }
  if (granularity === "hours") {
    const d = new Date(reading.recordedAt);
    const hour = String(d.getUTCHours()).padStart(2, "0");
    return locale === "es"
      ? `D${reading.missionDay} ${hour}:00`
      : `D${reading.missionDay} ${hour}:00`;
  }
  const time = formatIntradayAxis(reading.recordedAt, locale);
  return locale === "es" ? `D${reading.missionDay} ${time}` : `D${reading.missionDay} ${time}`;
}

function toChartRow(
  reading: VitalReadings,
  cumulative: number,
  granularity: ChartTimeGranularity,
  locale: AppLocale
): ChartTimelineRow {
  const dose = reading.radiationMsv ?? 0;
  return {
    axisLabel: axisLabelForGranularity(reading, granularity, locale),
    missionDay: reading.missionDay,
    recordedAt: reading.recordedAt,
    systolicBP: reading.systolicBP,
    heartRate: reading.heartRate,
    sleepHours: reading.sleepHours,
    bloodSugar: reading.bloodSugar,
    moodScore: reading.moodScore,
    radiationMsv: dose,
    radiationCumulative: Math.round(cumulative * 100) / 100,
    reading,
  };
}

export function buildChartTimeline(
  history: VitalReadings[],
  environment: MissionEnvironment,
  granularity: ChartTimeGranularity,
  locale: AppLocale
): { rows: ChartTimelineRow[]; source: VitalReadings[] } {
  const filtered = history.filter((h) => h.environment === environment);
  const source = filtered.length >= 10 ? filtered : buildMockMissionHistory(environment);

  const readings: VitalReadings[] =
    granularity === "days"
      ? source
      : source.flatMap((dayRow) =>
          buildIntradaySamples(dayRow, INTRADAY_SLOTS[granularity])
        );

  let cumulative = 0;
  const rows = readings.map((reading) => {
    cumulative += reading.radiationMsv ?? 0;
    return toChartRow(reading, cumulative, granularity, locale);
  });

  return { rows, source };
}

import type { AppLocale } from "@/lib/locale";

/** Mission-clock label with day + time (includes seconds for chart tooltips). */
export function formatMissionTimestamp(iso: string, locale: AppLocale): string {
  const d = new Date(iso);
  if (Number.isNaN(d.getTime())) return iso;
  return d.toLocaleString(locale === "es" ? "es-ES" : "en-US", {
    timeZone: "UTC",
    month: "short",
    day: "numeric",
    hour: "2-digit",
    minute: "2-digit",
    second: "2-digit",
    hour12: false,
  });
}

export function formatIntradayAxis(iso: string, locale: AppLocale): string {
  const d = new Date(iso);
  if (Number.isNaN(d.getTime())) return "";
  return d.toLocaleTimeString(locale === "es" ? "es-ES" : "en-US", {
    timeZone: "UTC",
    hour: "2-digit",
    minute: "2-digit",
    second: "2-digit",
    hour12: false,
  });
}

export function formatDayAxis(day: number, iso: string, locale: AppLocale): string {
  const time = formatIntradayAxis(iso, locale);
  return locale === "es" ? `D${day} ${time}` : `D${day} ${time}`;
}

export function parseMissionEventCodes(
  extra: Record<string, number | string> | undefined
): string[] {
  if (!extra?.missionEvents) return [];
  const raw = extra.missionEvents;
  if (typeof raw !== "string" || !raw.trim()) return [];
  return raw.split(",").map((s) => s.trim()).filter(Boolean);
}

# Product

<!-- impeccable:product-schema 1 -->

## Platform

web

## Stack

Next.js App Router, TypeScript, Tailwind CSS, daisyUI, lucide-react, recharts. Inferred from repository (NASA Space Apps 48h PMV).

## Users

Crew members and flight surgeons evaluating astronaut health during Artemis-style missions when ground support is delayed or unavailable. Secondary: judges and demo viewers at NASA Space Apps.

## Product Purpose

Offline-first mission health monitoring: log vitals, see body-system status, trends, and NASA HRP–inspired guidance without blocking on the network.

## Positioning

Clinical evaluation runs deterministically in the browser via `evaluateHealth()`; optional API sync does not gate the UX.

## Operating Context

Long-duration missions across station, spacecraft, rover, and lunar base environments with different latency and hazard profiles. Daily check-ins, scenario simulations for demos, bilingual ES/EN UI.

## Capabilities and Constraints

- Vitals form with streak and routine checklist; body map; timeline charts; rule engine; mock mission history and localStorage persistence.
- Semaphore statuses: normal / warning / critical (emerald, amber, rose).
- No fabricated NASA endorsements; educational copy from project data files.

## Brand Commitments

Product name in metadata: "Mission Health Control | NASA Space Apps". Dark mission-control aesthetic; clear simple English (with Spanish locale toggle).

## Evidence on Hand

Mock mission data (`src/data/mockMissionHistory.ts`), NASA hazard knowledge (`src/data/nasaKnowledge.ts`), no live crew telemetry.

## Product Principles

1. Offline-first for critical evaluation.
2. Types from `src/types/health.ts` as contract.
3. Explain why each measurement matters in space.
4. Lightweight stack—no heavy 3D or realtime DB for PMV.

## Accessibility & Inclusion

Skip link, live regions for status, keyboard-navigable body regions, focus-visible styling.

import type { MissionEnvironment } from "@/types/health";

function parseEnvironment(value: string | null): MissionEnvironment {
  if (
    value === "station" ||
    value === "spacecraft" ||
    value === "rover" ||
    value === "lunar_base" ||
    value === "mars_base"
  ) {
    return value;
  }
  return "station";
}

export async function GET(request: Request) {
  const { searchParams } = new URL(request.url);
  const environment = parseEnvironment(searchParams.get("environment"));
  const days = Math.min(
    30,
    Math.max(1, parseInt(searchParams.get("days") ?? "14", 10) || 14)
  );
  const seed = searchParams.get("seed") ?? undefined;
  const astronautId = searchParams.get("astronautId") ?? undefined;

  const feed = buildFakerMissionFeed(environment, days, seed, astronautId);

  return NextResponse.json({
    ok: true,
    source: "faker_mock_ground_api",
    ...feed,
    note: "Fictional ground API for demo sync; evaluateHealth() still runs fully on-device.",
  });
}

/** Shape telemetry to match roster archetype (last missions days). */
export function applyAstronautArchetype(
  telemetry: VitalReadings[],
  archetype: AstronautHealthArchetype,
  typicalScenarios: MissionScenarioId[]
): VitalReadings[] {
  if (telemetry.length === 0) return telemetry;
  const rows = telemetry.map((r) => ({ ...r }));
  const last = rows[rows.length - 1];
  const prev = rows.length > 1 ? rows[rows.length - 2] : last;

  switch (archetype) {
    case "healthy":
      last.systolicBP = Math.min(last.systolicBP, 122);
      last.heartRate = Math.min(last.heartRate, 78);
      last.stressLevel = Math.min(last.stressLevel ?? 5, 4);
      break;
    case "cardio_warning":
      last.systolicBP = 152;
      last.diastolicBP = 96;
      last.heartRate = 108;
      prev.systolicBP = 138;
      break;
    case "radiation_exposure":
      last.radiationMsv = 1.35;
      prev.radiationMsv = 0.95;
      last.heartRate = Math.max(last.heartRate, 96);
      last.extraMetrics = { ...last.extraMetrics, fatigueLevel: "elevated" };
      break;
    case "isolation_stress":
      last.sleepHours = 4.2;
      last.moodScore = 2;
      last.stressLevel = 8;
      last.extraMetrics = { ...last.extraMetrics, reactionTimeMs: 520 };
      break;
    case "metabolic_drift":
      last.bloodSugar = 158;
      last.temperature = 37.4;
      prev.bloodSugar = 132;
      break;
    case "multi_system_critical":
      last.systolicBP = 168;
      last.heartRate = 118;
      last.radiationMsv = 1.6;
      last.sleepHours = 3.5;
      last.stressLevel = 9;
      last.moodScore = 1;
      last.extraMetrics = {
        fatigueLevel: "elevated",
        reactionTimeMs: 580,
      };
      break;
  }

  last.extraMetrics = {
    ...last.extraMetrics,
    rosterArchetype: archetype,
    typicalScenarios: typicalScenarios.join(","),
  };

  return rows;
}

  if (intensity >= 8) return "#fb7185";
  if (intensity >= 5) return "#fbbf24";
  if (intensity >= 1) return "#34d399";
  return "#334155";
}

export function BodyMuscleChart({
  bodyState,
  ariaLabel,
  onMuscleActivate,
}: BodyMuscleChartProps) {
  const containerRef = useRef<HTMLDivElement>(null);
  const chartRef = useRef<BodyChart | null>(null);
  const onActivateRef = useRef(onMuscleActivate);

  useEffect(() => {
    onActivateRef.current = onMuscleActivate;
  }, [onMuscleActivate]);

  useEffect(() => {
    const el = containerRef.current;
    if (!el) return;

    chartRef.current = new BodyChart(el, {
      view: ViewSide.FRONT,
      bodyState,
      ariaLabel,
      showViewLabel: false,
      intensityColor: (value) => resolveHealthIntensityColor(value),
      className: "body-muscles-mission",
      onMuscleClick(id) {
        onActivateRef.current(id);
      },
    });

    return () => {
      chartRef.current?.destroy();
      chartRef.current = null;
    };
    // eslint-disable-next-line react-hooks/exhaustive-deps -- mount once; state via update()
  }, [ariaLabel]);

  useEffect(() => {
    chartRef.current?.update({ bodyState });
  }, [bodyState]);

import { MUSCLE_GROUPS, type MuscleGroup } from "@emmorts/body-muscles";
import type { BodyState, MuscleId } from "@emmorts/body-muscles";
import type { BodySystemStatus, HealthStatus, SystemStatusEntry } from "@/types/health";

const SYSTEM_GROUPS: Record<BodySystemStatus, MuscleGroup[]> = {
  neuro_behavioral: ["Head & Neck"],
  cardiovascular: ["Chest"],
  metabolic: ["Abdominals"],
  immune_musculoskeletal: ["Arms", "Legs", "Back", "Shoulders"],
};

const MUSCLE_TO_SYSTEM = new Map<MuscleId, BodySystemStatus>();

for (const [system, groups] of Object.entries(SYSTEM_GROUPS) as [
  BodySystemStatus,
  MuscleGroup[],
][]) {
  for (const group of groups) {
    for (const muscleId of MUSCLE_GROUPS[group]) {
      MUSCLE_TO_SYSTEM.set(muscleId, system);
    }
  }
}

export function muscleIdToBodySystem(id: MuscleId): BodySystemStatus | null {
  return MUSCLE_TO_SYSTEM.get(id) ?? null;
}

function statusToIntensity(status: HealthStatus): number {
  if (status === "critical") return 9;
  if (status === "warning") return 6;
  return 2;
}

export function buildBodyMuscleState(
  systemStatuses: SystemStatusEntry[],
  selectedSystem: BodySystemStatus
): BodyState {
  const statusBySystem = new Map<BodySystemStatus, HealthStatus>();
  for (const entry of systemStatuses) {
    statusBySystem.set(entry.system, entry.status);
  }

  const state: BodyState = {};
  for (const system of Object.keys(SYSTEM_GROUPS) as BodySystemStatus[]) {
    const intensity = statusToIntensity(statusBySystem.get(system) ?? "normal");
    const selected = system === selectedSystem;
    for (const group of SYSTEM_GROUPS[system]) {
      for (const muscleId of MUSCLE_GROUPS[group]) {
        state[muscleId] = { intensity, selected };
      }
    }
  }
  return state;
}

export async function fetchMissionFeed(
  environment: MissionEnvironment,
  days = 14,
  astronautId?: string
): Promise<MissionFeedResponse | null> {
  try {
    const params = new URLSearchParams({
      environment,
      days: String(days),
    });
    if (astronautId) params.set("astronautId", astronautId);
    const res = await fetch(`/api/mission-feed?${params.toString()}`, {

    if (!res.ok) return null;
    return (await res.json()) as MissionFeedResponse;
  } catch {
    return null;
  }
}

import {
  cardioDehydrationScenario,
  isolationDriftScenario,
  solarParticleScenario,
} from "@/data/mockMissionHistory";
import type { MissionEnvironment, MissionScenarioId, VitalReadings } from "@/types/health";

export const SCENARIO_APPLY_ORDER: MissionScenarioId[] = [
  "solar",
  "cardio",
  "isolation",
];

function scenarioFactory(
  id: MissionScenarioId
): (environment: MissionEnvironment) => VitalReadings {
  switch (id) {
    case "solar":
      return solarParticleScenario;
    case "cardio":
      return cardioDehydrationScenario;
    case "isolation":
      return isolationDriftScenario;
  }
}

function mergeVitalPair(base: VitalReadings, patch: VitalReadings): VitalReadings {
  return {
    ...base,
    systolicBP: Math.max(base.systolicBP, patch.systolicBP),
    diastolicBP: Math.max(base.diastolicBP, patch.diastolicBP),
    heartRate: Math.max(base.heartRate, patch.heartRate),
    bloodSugar: Math.max(base.bloodSugar, patch.bloodSugar),
    temperature: Math.max(base.temperature, patch.temperature),
    radiationMsv: Math.max(base.radiationMsv ?? 0, patch.radiationMsv ?? 0),
    sleepHours: Math.min(base.sleepHours, patch.sleepHours),
    moodScore: Math.min(base.moodScore, patch.moodScore),
    stressLevel: Math.max(base.stressLevel ?? 0, patch.stressLevel ?? 0),
    extraMetrics: { ...base.extraMetrics, ...patch.extraMetrics },
  };
}

/** Layer one or more mission scenarios onto a baseline reading (same mission day). */
export function mergeActiveScenarios(
  baseline: VitalReadings,
  active: MissionScenarioId[],
  environment: MissionEnvironment
): VitalReadings {
  let merged: VitalReadings = { ...baseline, environment };
  for (const id of SCENARIO_APPLY_ORDER) {
    if (!active.includes(id)) continue;
    const scenarioSample = scenarioFactory(id)(environment);
    merged = mergeVitalPair(merged, scenarioSample);
  }
  return {
    ...merged,
    missionDay: baseline.missionDay,
    environment,
    extraMetrics: {
      ...merged.extraMetrics,
      activeScenarios: active.join(","),
    },
  };
}

import type { MissionScenarioId, VitalReadings } from "@/types/health";

function hasSolarSignature(reading: VitalReadings): boolean {
  return (
    (reading.radiationMsv ?? 0) >= 1.4 &&
    reading.heartRate >= 100 &&
    (reading.extraMetrics?.fatigueLevel === "elevated" ||
      (reading.sleepHours ?? 8) < 6)
  );
}

function hasCardioSignature(reading: VitalReadings): boolean {
  return reading.systolicBP >= 155 && reading.heartRate >= 115;
}

function hasIsolationSignature(reading: VitalReadings): boolean {
  return (
    reading.sleepHours <= 4.5 &&
    (reading.stressLevel ?? 0) >= 7 &&
    (reading.moodScore ?? 5) <= 3
  );
}

export function detectMissionScenarios(reading: VitalReadings): MissionScenarioId[] {
  const active: MissionScenarioId[] = [];
  if (hasSolarSignature(reading)) active.push("solar");
  if (hasCardioSignature(reading)) active.push("cardio");
  if (hasIsolationSignature(reading)) active.push("isolation");
  return active;
}

/** @deprecated Use detectMissionScenarios */
export function detectMissionScenario(reading: VitalReadings): MissionScenarioId | null {
  const list = detectMissionScenarios(reading);
  return list[0] ?? null;
}

# NASA Space Apps — Health Monitoring MVP (48h team conventions)

## Team structure (3 developers)

Work inside **your** feature folder only. Do not edit another dev's feature folder without coordination.

| Developer | Folder | Responsibility |
|-----------|--------|----------------|
| Dev 1 | `src/features/body-map/`, `src/features/timeline/` | Interactive body map + mission charts |
| Dev 2 | `src/features/rules-engine/` | Deterministic offline health evaluation (NASA HRP–inspired thresholds) |
| Dev 3 | `src/features/vitals-form/` | Guided vitals input, streak widget, image preview |

Shared contracts (coordinate before changing):
- `src/types/health.ts` — single source of types for health data and evaluation
- `src/data/nasaKnowledge.ts` — educational copy and thresholds
- `src/data/mockMissionHistory.ts` — demo mission readings
- `src/app/page.tsx` — dashboard assembly only (thin shell)

## TypeScript

- **Strict mode** always. No `any`. Prefer explicit return types on exported functions.
- Import domain types from `@/types/health` (or `@/shared/types` re-export if present).
- New indicators: extend `extraMetrics` or update `health.ts` in a PR reviewed by all three.

## Architecture

- **Feature-based** under `src/features/<feature-name>/`.
- **Offline-first**: evaluation runs in the browser via `evaluateHealth()`. No blocking API for core UX.
- **Lightweight**: no heavy 3D, no ORM, no realtime DB for the PMV. `localStorage` for persistence optional.
- API routes are placeholders for future ground-sync only.

## UI language

- All user-facing text in **clear, simple English**. Do not assume medical training.
- Explain *why* a measurement matters in space in one short sentence when possible.

## Status color standard (Tailwind)

| Status | Semantics | Tailwind |
|--------|-----------|----------|
| `normal` | Within expected range | `emerald` (e.g. `text-emerald-400`, `bg-emerald-500/20`) |
| `warning` | Trend or value needs attention | `amber` |
| `critical` | Escalate to crew or ground | `rose` |

Use shared helpers from `@/lib/statusColors` when mapping `HealthStatus` to classes.

## Components

- Prefer small, named exports. One main component per file.
- Client components: `"use client"` only when needed (forms, charts, interactivity).
- Dark mission-control aesthetic: `bg-slate-950`, `text-slate-100`, subtle borders `border-slate-800`.

## Git / merge conflict avoidance

- Do not rename files in another feature folder.
- Add new files inside your folder; wire them from `page.tsx` in a single coordinated commit.

export interface FakerMissionFeed {
  feedId: string;
  generatedAt: string;
  environment: MissionEnvironment;
  days: number;
  crew: MockGroundCrew;
  signalLatencyMs: number;
  telemetry: VitalReadings[];

function seedFromString(value: string): number {
  let hash = 0;
  for (let i = 0; i < value.length; i++) {
    hash = (hash << 5) - hash + value.charCodeAt(i);
    hash |= 0;
  }
  return Math.abs(hash);
}

const LATENCY_MS: Record<MissionEnvironment, number> = {
  station: 500,
  spacecraft: 12 * 60 * 1000,
  rover: 2600,
  lunar_base: 2600,
  mars_base: 18 * 60 * 1000,
};

const HABITAT_BY_ENV: Record<MissionEnvironment, string[]> = {
  station: ["ISS USOS", "Columbus", "Kibo"],
  spacecraft: ["Orion CM", "Deep Space Transport", "Gateway HALO"],
  rover: ["Artemis LTV", "PRESS Lunar Rover", "Mars Science Rover"],
  lunar_base: ["Artemis South Pole Habitat", "Lunar Gateway Logistics"],
  mars_base: ["Mars Ice Home", "Olympus Mons Camp", "Hellas Planitia Lab"],
};

/** Server-only synthetic ground feed — seeds Faker, layers noise on HRP-shaped baselines. */
export function buildFakerMissionFeed(
  environment: MissionEnvironment,
  days: number,
  seed?: string,
  astronautId?: string
): FakerMissionFeed {
  const rosterEntry = astronautId ? getAstronautById(astronautId) : undefined;
  /** User-selected mission context; do not override with crew home habitat. */
  const env = environment;
  const seedValue = seed ?? `${astronautId ?? "crew"}-${env}-${days}`;
  faker.seed(seedFromString(seedValue));

  let baseline = buildMockMissionHistory(env).slice(0, days).map((row) => ({
    ...row,
    environment: env,
    systolicBP: row.systolicBP + faker.number.int({ min: -3, max: 3 }),
    diastolicBP: row.diastolicBP + faker.number.int({ min: -2, max: 2 }),
    heartRate: row.heartRate + faker.number.int({ min: -4, max: 4 }),
    bloodSugar: row.bloodSugar + faker.number.int({ min: -5, max: 5 }),
    temperature:
      Math.round((row.temperature + faker.number.float({ min: -0.15, max: 0.15 })) * 10) /
      10,
    sleepHours:
      Math.round((row.sleepHours + faker.number.float({ min: -0.3, max: 0.3 })) * 10) / 10,
    recordedAt: row.recordedAt,
  }));

  if (rosterEntry) {
    baseline = applyAstronautArchetype(
      baseline,
      rosterEntry.archetype,
      rosterEntry.typicalScenarios
    );
    const last = baseline[baseline.length - 1];
    if (last) {
      last.extraMetrics = {
        ...last.extraMetrics,
        astronautId: rosterEntry.id,
      };
    }
  }

  const crewName = rosterEntry?.fullName ?? faker.person.fullName();
  const callSign = rosterEntry?.callSign ?? `Artemis-${faker.number.int({ min: 410, max: 420 })}`;

  return {
    feedId: faker.string.uuid(),
    generatedAt: new Date().toISOString(),
    environment: env,
    days,
    signalLatencyMs: LATENCY_MS[env],
    rosterSize: ASTRONAUT_ROSTER.length,
    crew: {
      callSign,
      commander: crewName,
      medicalOfficer: rosterEntry?.role ?? faker.person.fullName(),
      agency: rosterEntry?.agency ?? faker.helpers.arrayElement(["NASA", "CSA", "ESA", "JAXA"]),
      habitatModule: faker.helpers.arrayElement(HABITAT_BY_ENV[env]),
      astronautId: rosterEntry?.id,
      archetype: rosterEntry?.archetype,
    },
    telemetry: baseline,
  };
}

# Mission Health Control

Offline-first astronaut health monitoring demo for **NASA Space Apps Challenge**. The browser runs deterministic evaluation inspired by NASA Human Research Program (HRP) thresholds—no blocking API required for core UX.

## Features

- **Mission dashboard** — crew profile, communication latency, and environment-specific hazards
- **Body map** — status by body system with linked NASA hazard education
- **Mission timeline charts** — 14-day cardiovascular, recovery, and radiation trends
- **Vitals check-in** — guided daily log with optional camera/ultrasound attachment
- **Rules engine** — `evaluateHealth()` with escalation: self-care → crew → ground flight surgeon
- **Mission scenarios** — solar particles, cardio/dehydration, isolation (stackable simulations)

## Stack

Next.js (App Router), TypeScript, Tailwind CSS, daisyUI, `lucide-react`, Recharts.

import { NextResponse } from "next/server";
import { buildMockMissionHistory } from "@/data/mockMissionHistory";
import type { MissionEnvironment, VitalReadings } from "@/types/health";

/** In-memory stub for ground-sync demos — resets on deploy (Render ephemeral FS). */
const memoryLogs: VitalReadings[] = [];

function parseEnvironment(value: string | null): MissionEnvironment {
  if (
    value === "station" ||
    value === "spacecraft" ||
    value === "rover" ||
    value === "lunar_base" ||
    value === "mars_base"
  ) {
    return value;
  }
  return "station";
}

function simulateLatencyMs(environment: MissionEnvironment): number {
  const map: Record<MissionEnvironment, number> = {
    station: 500,
    spacecraft: 12 * 60 * 1000,
    rover: 2600,
    lunar_base: 2600,
    mars_base: 18 * 60 * 1000,
  };
  return map[environment];
}

export async function GET(request: Request) {
  const { searchParams } = new URL(request.url);
  const environment = parseEnvironment(searchParams.get("environment"));
  const days = Math.min(
    30,
    Math.max(1, parseInt(searchParams.get("days") ?? "14", 10) || 14)
  );

  if (searchParams.get("environment") || searchParams.get("days")) {
    const synthetic = buildMockMissionHistory(environment).slice(0, days);
    return NextResponse.json({
      ok: true,
      mode: "synthetic_telemetry",
      environment,
      days,
      logs: synthetic,
      signalLatencyMs: simulateLatencyMs(environment),
      note: "Synthetic telemetry for demos; client evaluation remains offline-first.",
    });
  }

  return NextResponse.json({
    ok: true,
    count: memoryLogs.length,
    logs: memoryLogs,
    note: "Client-side evaluation works fully offline; this endpoint is optional.",
  });
}

interface PostPayload extends VitalReadings {
  syncAck?: boolean;
}

export async function POST(request: Request) {
  try {
    const body = (await request.json()) as PostPayload;
    if (typeof body.missionDay !== "number" || !body.recordedAt) {
      return NextResponse.json(
        { ok: false, error: "Invalid vitals payload" },
        { status: 400 }
      );
    }

    const { syncAck, ...vitals } = body;
    memoryLogs.push(vitals);

    const latencyMs = simulateLatencyMs(vitals.environment ?? "station");
    await new Promise((resolve) => setTimeout(resolve, Math.min(latencyMs, 1500)));

    return NextResponse.json({
      ok: true,
      received: vitals.missionDay,
      groundSyncAck: syncAck === true,
      simulatedSignalLatencyMs: latencyMs,
      message:
        "Ground control acknowledged receipt (simulated). Clinical rules already ran on-device.",
    });
  } catch {
    return NextResponse.json({ ok: false, error: "Bad JSON" }, { status: 400 });
  }
}

import type { MissionEnvironment, VitalReadings } from "@/types/health";
import { missionLogEventsForDay } from "@/data/intradaySamples";

const baseDate = new Date("2026-11-01T00:00:00Z");

function entry(
  day: number,
  partial: Partial<VitalReadings> & Pick<VitalReadings, "environment">
): VitalReadings {
  const d = new Date(baseDate);
  d.setUTCDate(d.getUTCDate() + day - 1);
  /** Stagger daily rollup time (hours:minutes:seconds UTC) per mission day. */
  d.setUTCHours(6 + (day % 12), (day * 7) % 60, (day * 11) % 60, 0);
  const env = partial.environment;
  return {
    missionDay: day,
    recordedAt: d.toISOString(),
    systolicBP: 118,
    diastolicBP: 76,
    heartRate: 72,
    bloodSugar: 98,
    temperature: 36.7,
    sleepHours: 7.2,
    moodScore: 4,
    stressLevel: 3,
    radiationMsv: 0.4 + day * 0.02,
    ...partial,
    environment: env,
    extraMetrics: {
      ...(partial.extraMetrics ?? {}),
      missionEvents: missionLogEventsForDay(day, env),
    },
  };
}

/** 14 mission days — subtle upward BP / downward sleep trend after day 9 */
export function buildMockMissionHistory(
  environment: MissionEnvironment = "station"
): VitalReadings[] {
  const env = environment;
  const days: VitalReadings[] = [];

  for (let day = 1; day <= 14; day++) {
    let systolic = 112 + Math.sin(day / 3) * 4;
    let diastolic = 74;
    let heartRate = 68 + (day % 3);
    let sleep = 7.4 - (day > 10 ? 0.15 * (day - 10) : 0);
    let mood = day > 11 ? 3 : 4;
    let glucose = 95 + (day > 8 ? (day - 8) * 2 : 0);
    let stress = day > 12 ? 5 : 3;

    if (day === 12) {
      systolic = 138;
      diastolic = 92;
      heartRate = 98;
    }
    if (day === 13) {
      sleep = 4.5;
      mood = 2;
      stress = 6;
    }
    if (day === 14) {
      systolic = 142;
      glucose = 165;
    }

    const envDoseBoost =
      env === "mars_base" ? 0.12 : env === "spacecraft" ? 0.08 : env === "rover" ? 0.05 : 0;
    const radiationMsv = day >= 14 ? 0.85 + envDoseBoost : 0.35 + day * 0.03 + envDoseBoost;

    days.push(
      entry(day, {
        environment: env,
        systolicBP: Math.round(systolic),
        diastolicBP: Math.round(diastolic),
        heartRate: Math.round(heartRate),
        bloodSugar: Math.round(glucose),
        temperature: 36.5 + (day === 13 ? 0.4 : 0.2),
        sleepHours: Math.round(sleep * 10) / 10,
        moodScore: mood,
        stressLevel: stress,
        radiationMsv,
      })
    );
  }

  return days;
}

import { EXAM_THRESHOLDS, NASA_HAZARDS_KNOWLEDGE } from "@/data/nasaKnowledge";
import { EVALUATOR_COPY, ENVIRONMENT_MISSION_META, type AppLocale } from "@/lib/locale";
import type {
  ActionableGuidance,
  BodySystemStatus,
  EscalationLevel,
  HealthEvaluation,
  HealthStatus,
  MissionEnvironment,
  SpaceflightHazard,
  SystemStatusEntry,
  VitalReadings,
} from "@/types/health";

function radiationStatusForEnvironment(
  dose: number,
  environment: MissionEnvironment
): HealthStatus {
  const deepSpace =
    environment === "spacecraft" || environment === "mars_base";
  const warningAt = deepSpace ? 0.85 : 1.0;
  const criticalAt = deepSpace ? 1.2 : 1.5;
  if (dose > criticalAt) return "critical";
  if (dose > warningAt) return "warning";
  return "normal";
}

function scoreMetric(
  value: number,
  normal: { min?: number; max?: number },
  warning: { min?: number; max?: number }
): HealthStatus {
  const inRange = (range: { min?: number; max?: number }, v: number) => {
    if (range.min !== undefined && v < range.min) return false;
    if (range.max !== undefined && v > range.max) return false;
    return true;
  };
  if (inRange(normal, value)) return "normal";
  if (inRange(warning, value)) return "warning";
  return "critical";
}

function trendDelta(recent: VitalReadings[], pick: (v: VitalReadings) => number): number {
  if (recent.length < 2) return 0;
  const sorted = [...recent].sort((a, b) => a.missionDay - b.missionDay);
  const last = sorted[sorted.length - 1];
  const prev = sorted[sorted.length - 2];
  return pick(last) - pick(prev);
}

function worst(a: HealthStatus, b: HealthStatus): HealthStatus {
  const rank: Record<HealthStatus, number> = {
    normal: 0,
    warning: 1,
    critical: 2,
  };
  return rank[a] >= rank[b] ? a : b;
}

function systemForMetric(metric: string): BodySystemStatus {
  const map: Record<string, BodySystemStatus> = {
    systolicBP: "cardiovascular",
    diastolicBP: "cardiovascular",
    heartRate: "cardiovascular",
    bloodSugar: "metabolic",
    temperature: "metabolic",
    sleepHours: "neuro_behavioral",
    moodScore: "neuro_behavioral",
    stressLevel: "neuro_behavioral",
    radiationMsv: "immune_musculoskeletal",
  };
  return map[metric] ?? "immune_musculoskeletal";
}

function buildGuidance(
  hazards: Set<SpaceflightHazard>,
  actions: string[],
  why: string
): ActionableGuidance {
  const hazardImpact = NASA_HAZARDS_KNOWLEDGE.filter((h) =>
    hazards.has(h.hazard)
  ).map((h) => h.hazard);
  return {
    whyNeededInSpace: why,
    hazardImpact,
    recommendedActions: actions,
  };
}

function reactionTimeMs(current: VitalReadings): number | undefined {
  const raw = current.extraMetrics?.reactionTimeMs;
  if (typeof raw === "number") return raw;
  if (typeof raw === "string") {
    const n = Number(raw);
    return Number.isFinite(n) ? n : undefined;
  }
  return undefined;
}

/**
 * Pure offline evaluator — Dev 2 extends thresholds and escalation rules here.
 */
export function evaluateHealth(
  current: VitalReadings,
  recentHistory: VitalReadings[] = [],
  locale: AppLocale = "en"
): HealthEvaluation {
  const copy = EVALUATOR_COPY[locale];
  const environment = current.environment;
  const envMeta = ENVIRONMENT_MISSION_META[locale][environment];
  const history = [...recentHistory, current].sort(
    (a, b) => a.missionDay - b.missionDay
  );

  const metricStatuses: Record<string, HealthStatus> = {};
  for (const th of EXAM_THRESHOLDS) {
    if (th.metric === "radiationMsv" || th.metric === "stressLevel") continue;
    const value = current[th.metric] as number;
    metricStatuses[th.metric] = scoreMetric(value, th.normal, th.warning);
  }

  if (current.radiationMsv !== undefined) {
    const st = radiationStatusForEnvironment(current.radiationMsv, environment);
    metricStatuses.radiationMsv = st;
    metricStatuses.radiation = st;
  }

  if (current.stressLevel !== undefined) {
    const stress = current.stressLevel;
    let st: HealthStatus = "normal";
    if (stress >= 8) st = "critical";
    else if (stress >= 6) st = "warning";
    metricStatuses.stressLevel = st;
  }

  const reactionMs = reactionTimeMs(current);
  if (reactionMs !== undefined) {
    let st: HealthStatus = "normal";
    if (reactionMs > 650) st = "critical";
    else if (reactionMs > 450) st = "warning";
    metricStatuses.reactionTimeMs = st;
  }

  const bpTrend = trendDelta(history, (v) => v.systolicBP);
  const sleepTrend = trendDelta(history, (v) => v.sleepHours);
  if (bpTrend >= 8) {
    metricStatuses.systolicBP = worst(metricStatuses.systolicBP, "warning");
  }
  if (sleepTrend <= -1.5) {
    metricStatuses.sleepHours = worst(metricStatuses.sleepHours, "warning");
  }

  const bySystem = new Map<BodySystemStatus, HealthStatus>();
  for (const th of EXAM_THRESHOLDS) {
    if (th.metric === "radiationMsv" || th.metric === "stressLevel") continue;
    const sys = systemForMetric(th.metric);
    const st = metricStatuses[th.metric];
    bySystem.set(sys, worst(bySystem.get(sys) ?? "normal", st));
  }
  if (metricStatuses.radiation === "warning" || metricStatuses.radiation === "critical") {
    bySystem.set(
      "immune_musculoskeletal",
      worst(bySystem.get("immune_musculoskeletal") ?? "normal", metricStatuses.radiation)
    );
  }
  if (metricStatuses.stressLevel) {
    bySystem.set(
      "neuro_behavioral",
      worst(bySystem.get("neuro_behavioral") ?? "normal", metricStatuses.stressLevel)
    );
  }
  if (metricStatuses.reactionTimeMs) {
    bySystem.set(
      "neuro_behavioral",
      worst(bySystem.get("neuro_behavioral") ?? "normal", metricStatuses.reactionTimeMs)
    );
  }

  const systemStatuses: SystemStatusEntry[] = (
    [
      "cardiovascular",
      "neuro_behavioral",
      "metabolic",
      "immune_musculoskeletal",
    ] as const
  ).map((system) => ({
    system,
    status: bySystem.get(system) ?? "normal",
    summary:
      (bySystem.get(system) ?? "normal") === "normal"
        ? copy.systemNormal
        : copy.systemWarning,
  }));

  let overallStatus: HealthStatus = "normal";
  for (const st of Object.values(metricStatuses)) {
    overallStatus = worst(overallStatus, st);
  }
  for (const s of systemStatuses) {
    overallStatus = worst(overallStatus, s.status);
  }

  let escalationLevel: EscalationLevel = "self_care";
  if (overallStatus === "warning") escalationLevel = "crew_attention";
  if (overallStatus === "critical") escalationLevel = "ground_flight_surgeon";
  if (
    (environment === "spacecraft" || environment === "mars_base") &&
    metricStatuses.radiation === "warning"
  ) {
    escalationLevel = worstEscalation(escalationLevel, "crew_attention");
  }
  if (
    current.systolicBP >= 160 ||
    current.heartRate >= 120 ||
    current.temperature >= 38.2
  ) {
    escalationLevel = "ground_flight_surgeon";
    overallStatus = "critical";
  }
  if (
    (current.stressLevel !== undefined && current.stressLevel >= 8) ||
    (metricStatuses.reactionTimeMs === "warning" ||
      metricStatuses.reactionTimeMs === "critical")
  ) {
    escalationLevel = worstEscalation(escalationLevel, "crew_attention");
    if (
      current.stressLevel !== undefined &&
      current.stressLevel >= 8 &&
      (metricStatuses.sleepHours === "critical" ||
        metricStatuses.moodScore === "critical")
    ) {
      escalationLevel = worstEscalation(escalationLevel, "crew_attention");
    }
  }

  const guidance: ActionableGuidance[] = [];

  if (
    metricStatuses.systolicBP !== "normal" ||
    metricStatuses.heartRate !== "normal"
  ) {
    guidance.push(
      buildGuidance(
        new Set(["gravity", "distance"]),
        copy.guidance.cardio.actions,
        copy.guidance.cardio.why
      )
    );
  }

  if (
    metricStatuses.sleepHours !== "normal" ||
    metricStatuses.moodScore !== "normal" ||
    (metricStatuses.stressLevel && metricStatuses.stressLevel !== "normal") ||
    (metricStatuses.reactionTimeMs && metricStatuses.reactionTimeMs !== "normal")
  ) {
    guidance.push(
      buildGuidance(
        new Set(["isolation", "closed_environment"]),
        copy.guidance.neuro.actions,
        copy.guidance.neuro.why
      )
    );
  }

  if (metricStatuses.bloodSugar !== "normal" || metricStatuses.temperature !== "normal") {
    guidance.push(
      buildGuidance(
        new Set(["closed_environment", "distance"]),
        copy.guidance.metabolic.actions,
        copy.guidance.metabolic.why
      )
    );
  }

  if (metricStatuses.radiation === "warning" || metricStatuses.radiation === "critical") {
    guidance.push(
      buildGuidance(
        new Set(["radiation"]),
        [
          locale === "es"
            ? "Refugio en módulo blindado y repetir dosimetría."
            : "Shelter in shielded module and repeat dosimetry.",
          locale === "es"
            ? "Informar al oficial médico y registrar fatiga."
            : "Inform crew medical officer and log fatigue.",
        ],
        locale === "es"
          ? "Picos de radiación afectan inmunidad y requieren contramedidas inmediatas."
          : "Radiation spikes affect immunity and need immediate countermeasures.",
      )
    );
  }

  if (guidance.length === 0) {
    guidance.push(
      buildGuidance(
        new Set(["distance"]),
        copy.guidance.default.actions,
        copy.guidance.default.why
      )
    );
  }

  if (environment === "spacecraft" || environment === "mars_base") {
    guidance.push(
      buildGuidance(
        new Set(["distance", "radiation"]),
        locale === "es"
          ? [
              "Actuar de forma autónoma: el consultor en Tierra recibe datos con retraso (" +
                envMeta.latencyLabel +
                ").",
              "Documentar síntomas neuro-ocular (SANS) y escalar localmente si la visión cambia.",
            ]
          : [
              "Act autonomously: ground consult is delayed (" + envMeta.latencyLabel + ").",
              "Log neuro-ocular SANS symptoms and escalate locally if vision shifts.",
            ],
        locale === "es"
          ? "En tránsito profundo o en Marte la señal tarda minutos; las decisiones urgentes deben seguir reglas HRP a bordo."
          : "Deep transit or Mars link delay is measured in minutes; urgent calls follow onboard HRP rules.",
      )
    );
  }

  if (environment === "rover" || environment === "lunar_base") {
    guidance.push(
      buildGuidance(
        new Set(["gravity", "closed_environment"]),
        locale === "es"
          ? [
              "Chequeo post-EVA: hidratación, irritación por polvo lunar y estrés térmico del traje.",
              "Registrar presión arterial y FC 30 min tras reingreso al hábitat.",
            ]
          : [
              "Post-EVA check: hydration, lunar dust irritation, and suit thermal stress.",
              "Re-check blood pressure and heart rate 30 min after re-entry.",
            ],
        locale === "es"
          ? "En superficie, el polvo y la gravedad parcial amplifican deshidratación y fatiga tras cada salida."
          : "On the surface, dust and partial gravity amplify dehydration and fatigue after each EVA.",
      )
    );
  }

  if (environment === "station") {
    guidance.push(
      buildGuidance(
        new Set(["closed_environment", "gravity"]),
        locale === "es"
          ? [
              "Monitorizar CO₂ de cabina y síntomas de hipoxia leve (cefalea, somnolencia).",
              "Anotar cambios de visión asociados a SANS y fluidos en microgravedad.",
            ]
          : [
              "Track cabin CO₂ and mild hypoxia signs (headache, drowsiness).",
              "Log vision shifts linked to SANS and fluid redistribution in microgravity.",
            ],
        locale === "es"
          ? "La estación combina ambiente cerrado y microgravedad: CO₂ y SANS son riesgos HRP frecuentes."
          : "The station mixes closed ecology and microgravity; CO₂ and SANS are common HRP risks.",
      )
    );
  }

  const headline =
    escalationLevel === "ground_flight_surgeon"
      ? copy.headlines.ground
      : escalationLevel === "crew_attention"
        ? copy.headlines.crew
        : copy.headlines.ok;

  return {
    overallStatus,
    escalationLevel,
    systemStatuses,
    guidance,
    headline,
  };
}

function worstEscalation(a: EscalationLevel, b: EscalationLevel): EscalationLevel {
  const rank: Record<EscalationLevel, number> = {
    self_care: 0,
    crew_attention: 1,
    ground_flight_surgeon: 2,
  };
  return rank[a] >= rank[b] ? a : b;
}

"use client";

import { useCallback, useEffect, useRef, useState } from "react";
import { CheckSquare, Zap } from "lucide-react";
import type { AppLocale } from "@/lib/locale";
import { UI_STRINGS } from "@/lib/locale";

const CHECKLIST_KEY = "healthnasa_daily_checklist";
const POINTS_KEY = "healthnasa_readiness_points";
const REACTION_KEY = "healthnasa_reaction_avg_ms";

const CHECKLIST_IDS = [
  "resistive",
  "hydration",
  "sleep",
  "radiation",
  "cabin",
] as const;

type ChecklistId = (typeof CHECKLIST_IDS)[number];

interface MissionRoutinePanelProps {
  locale?: AppLocale;
  onReactionRecorded?: (averageMs: number) => void;
}

export function MissionRoutinePanel({
  locale = "en",
  onReactionRecorded,
}: MissionRoutinePanelProps) {
  const ui = UI_STRINGS[locale];

  const checklistLabels: Record<ChecklistId, string> = {
    resistive: ui.checklistResistiveExercise,
    hydration: ui.checklistHydration,
    sleep: ui.checklistSleep,
    radiation: ui.checklistRadiation,
    cabin: ui.checklistCabin,
  };

  const emptyChecklist = (): Record<ChecklistId, boolean> => ({
    resistive: false,
    hydration: false,
    sleep: false,
    radiation: false,
    cabin: false,
  });

  const [checked, setChecked] =
    useState<Record<ChecklistId, boolean>>(emptyChecklist);
  const [points, setPoints] = useState(0);
  const [reactionAvg, setReactionAvg] = useState<number | null>(null);

  useEffect(() => {
    try {
      const base = emptyChecklist();
      const raw = localStorage.getItem(CHECKLIST_KEY);
      if (raw) {
        setChecked({ ...base, ...(JSON.parse(raw) as Record<ChecklistId, boolean>) });
      }
      const pts = localStorage.getItem(POINTS_KEY);
      if (pts) setPoints(parseInt(pts, 10) || 0);
      const rx = localStorage.getItem(REACTION_KEY);
      if (rx) setReactionAvg(parseInt(rx, 10) || null);
    } catch {
      /* offline / private mode */
    }
  }, []);

  const [reactionPhase, setReactionPhase] = useState<
    "idle" | "waiting" | "go" | "done"
  >("idle");
  const [trial, setTrial] = useState(0);
  const [trialTimes, setTrialTimes] = useState<number[]>([]);
  const goAtRef = useRef<number | null>(null);
  const waitTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);

  const persistChecklist = useCallback(
    (next: Record<ChecklistId, boolean>, nextPoints: number) => {
      try {
        localStorage.setItem(CHECKLIST_KEY, JSON.stringify(next));
        localStorage.setItem(POINTS_KEY, String(nextPoints));
      } catch {
        /* ignore */
      }
    },
    []
  );

  function toggleItem(id: ChecklistId) {
    setChecked((prev) => {
      const wasOn = prev[id];
      const next = { ...prev, [id]: !wasOn };
      const delta = wasOn ? -10 : 10;
      setPoints((p) => {
        const nextPoints = Math.max(0, p + delta);
        persistChecklist(next, nextPoints);
        return nextPoints;
      });
      return next;
    });
  }

  function startReactionTest() {
    if (waitTimerRef.current) clearTimeout(waitTimerRef.current);
    setTrial(0);
    setTrialTimes([]);
    setReactionPhase("waiting");
    scheduleGoSignal();
  }

  function scheduleGoSignal() {
    const delay = 1200 + Math.random() * 2200;
    waitTimerRef.current = setTimeout(() => {
      goAtRef.current = performance.now();
      setReactionPhase("go");
    }, delay);
  }

  function onReactionTap() {
    if (reactionPhase === "waiting") {
      if (waitTimerRef.current) clearTimeout(waitTimerRef.current);
      setReactionPhase("idle");
      return;
    }
    if (reactionPhase !== "go" || goAtRef.current === null) return;

    const elapsed = Math.round(performance.now() - goAtRef.current);
    const nextTimes = [...trialTimes, elapsed];
    setTrialTimes(nextTimes);
    goAtRef.current = null;

    if (nextTimes.length >= 3) {
      const avg = Math.round(
        nextTimes.reduce((a, b) => a + b, 0) / nextTimes.length
      );
      setReactionAvg(avg);
      setReactionPhase("done");
      try {
        localStorage.setItem(REACTION_KEY, String(avg));
      } catch {
        /* ignore */
      }
      onReactionRecorded?.(avg);
    } else {
      setTrial(nextTimes.length);
      setReactionPhase("waiting");
      scheduleGoSignal();
    }
  }

  const completedCount = CHECKLIST_IDS.filter((id) => checked[id]).length;
  const progressPct = (completedCount / CHECKLIST_IDS.length) * 100;

  const targetColor =
    reactionPhase === "go"
      ? "bg-emerald-500 border-emerald-300"
      : reactionPhase === "waiting"
        ? "bg-amber-500/40 border-amber-400"
        : "bg-slate-700 border-slate-600";

  return (
    <div className="mission-panel p-4">
      <div className="flex items-center justify-between text-sm text-base-content/80">
        <span>{ui.readinessPoints}</span>
        <span className="font-medium text-success">{points}</span>
      </div>
      <progress
        className="progress progress-success mt-2 w-full"
        value={progressPct}
        max={100}
        aria-label={`${ui.readinessPoints}: ${completedCount} ${ui.of} ${CHECKLIST_IDS.length}`}
      />
      <ul className="mt-4 space-y-2">
        {CHECKLIST_IDS.map((id) => (
          <li key={id}>
            <label className="flex cursor-pointer items-center gap-2 text-sm">
              <input
                type="checkbox"
                checked={checked[id]}
                onChange={() => toggleItem(id)}
                className="checkbox checkbox-sm checkbox-success"
              />
              <CheckSquare className="h-4 w-4 text-base-content/60" aria-hidden />
              {checklistLabels[id]}
            </label>
          </li>
        ))}
      </ul>

      <div className="divider my-6" />

      <h4 className="flex items-center gap-2 text-base font-semibold">
        <Zap className="h-4 w-4 text-primary" aria-hidden />
        {ui.reactionTest}
      </h4>
      <p className="mt-1 text-sm text-base-content/75">{ui.reactionTestHint}</p>
      <div className="mt-3 flex flex-wrap items-center gap-3">
        {reactionPhase === "idle" || reactionPhase === "done" ? (
          <button type="button" onClick={startReactionTest} className="btn btn-primary btn-sm">
            {ui.reactionStart}
          </button>
        ) : (
          <button
            type="button"
            onClick={onReactionTap}
            aria-label={
              reactionPhase === "waiting" ? ui.reactionWait : ui.reactionTap
            }
            className={`btn flex h-20 w-20 min-h-20 items-center justify-center rounded-box border-2 text-xs ${targetColor}`}
          >
            {reactionPhase === "waiting"
              ? ui.reactionWait
              : ui.reactionTap}
          </button>
        )}
        {trial > 0 && reactionPhase !== "done" && (
          <span className="text-xs text-base-content/75">
            {ui.stepOf} {trial + 1} {ui.of} 3
          </span>
        )}
        {reactionAvg !== null && (
          <p className="text-sm" role="status" aria-live="polite">
            {ui.reactionResult}: <strong>{reactionAvg}</strong>
          </p>
        )}
      </div>
    </div>
  );
}

"use client";

import { useCallback, useMemo, useState } from "react";
import type { MuscleId } from "@emmorts/body-muscles";
import { NASA_HAZARDS_KNOWLEDGE } from "@/data/nasaKnowledge";
import {
  BODY_REGION_COPY,
  HAZARD_DISPLAY,
  statusLabelForLocale,
  type AppLocale,
  UI_STRINGS,
} from "@/lib/locale";
import { statusStyles } from "@/lib/statusColors";
import type { BodySystemStatus, HealthStatus, SystemStatusEntry } from "@/types/health";
import { BodyMuscleChart } from "./BodyMuscleChart";
import { buildBodyMuscleState, muscleIdToBodySystem } from "./buildBodyMuscleState";

type RegionId = BodySystemStatus;

const REGION_IDS: RegionId[] = [
  "neuro_behavioral",
  "cardiovascular",
  "metabolic",
  "immune_musculoskeletal",
];

interface BodyMapProps {
  systemStatuses: SystemStatusEntry[];
  locale?: AppLocale;
}

export function BodyMap({ systemStatuses, locale = "es" }: BodyMapProps) {
  const ui = UI_STRINGS[locale];
  const regions = BODY_REGION_COPY[locale];

  const statusBySystem = useMemo(() => {
    const m = new Map<BodySystemStatus, HealthStatus>();
    for (const s of systemStatuses) m.set(s.system, s.status);
    return m;
  }, [systemStatuses]);

  const [selected, setSelected] = useState<RegionId>("cardiovascular");
  const regionCopy = regions[selected];
  const status = statusBySystem.get(selected) ?? "normal";
  const styles = statusStyles(status);

  const bodyState = useMemo(
    () => buildBodyMuscleState(systemStatuses, selected),
    [systemStatuses, selected]
  );

  const relatedHazards = NASA_HAZARDS_KNOWLEDGE.filter((h) =>
    h.relatedSystems.includes(selected)
  );

  const onMuscleActivate = useCallback((id: MuscleId) => {
    const system = muscleIdToBodySystem(id);
    if (system) setSelected(system);
  }, []);

  function selectRegion(id: RegionId) {
    setSelected(id);
  }

  function onRegionKeyDown(
    event: React.KeyboardEvent<HTMLButtonElement>,
    id: RegionId
  ) {
    const index = REGION_IDS.indexOf(id);
    if (index < 0) return;
    if (event.key === "ArrowRight" || event.key === "ArrowDown") {
      event.preventDefault();
      selectRegion(REGION_IDS[(index + 1) % REGION_IDS.length]);
    } else if (event.key === "ArrowLeft" || event.key === "ArrowUp") {
      event.preventDefault();
      selectRegion(REGION_IDS[(index - 1 + REGION_IDS.length) % REGION_IDS.length]);
    }
  }

  return (
    <div className="grid gap-4 lg:grid-cols-2">
      <div className="mission-panel p-4">
        <div
          role="radiogroup"
          aria-label={ui.bodyRegionPicker}
          className="mb-3 flex flex-wrap gap-2"
        >
          {REGION_IDS.map((id) => {
            const st = statusBySystem.get(id) ?? "normal";
            const regionLabel = regions[id].label;
            const statusText = statusLabelForLocale(locale, st);
            const isSelected = selected === id;
            const stStyles = statusStyles(st);
            return (
              <button
                key={id}
                type="button"
                role="radio"
                aria-checked={isSelected}
                aria-label={`${regionLabel}, ${statusText}${
                  isSelected ? `, ${ui.bodyRegionSelected}` : ""
                }`}
                onClick={() => selectRegion(id)}
                onKeyDown={(e) => onRegionKeyDown(e, id)}
                className={`btn btn-xs sm:btn-sm gap-2 ${
                  isSelected
                    ? `btn-primary ring-2 ring-offset-2 ring-offset-base-200 ${stStyles.ring}`
                    : "btn-outline"
                }`}
              >
                <span
                  className={`h-2 w-2 rounded-full ${stStyles.dot}`}
                  aria-hidden
                />
                {regionLabel}
              </button>
            );
          })}
        </div>

        <BodyMuscleChart
          bodyState={bodyState}
          ariaLabel={ui.bodyMapAria}
          onMuscleActivate={onMuscleActivate}
        />
        <p className="mt-3 text-center text-sm text-base-content/75">{ui.bodyMapHint}</p>
      </div>

      <div
        className={`mission-panel p-4 ring-2 ${styles.ring} ${styles.bg}`}
        aria-live="polite"
      >
        <div className="flex items-center gap-2">
          <span className={`h-2.5 w-2.5 rounded-full ${styles.dot}`} aria-hidden />
          <h3 className="text-lg font-semibold text-base-content">{regionCopy.label}</h3>
          <span className={`ml-auto text-sm font-semibold ${styles.text}`}>
            {statusLabelForLocale(locale, status)}
          </span>
        </div>
        <p className="mt-2 text-sm text-base-content/85">{regionCopy.summary}</p>
        <p className="mt-3 text-xs font-medium uppercase tracking-wide text-base-content/70">
          {ui.relatedHazards}
        </p>
        <ul className="mt-2 space-y-4 text-sm text-base-content/85">
          {relatedHazards.map((h) => {
            const display = HAZARD_DISPLAY[locale][h.hazard];
            return (
              <li key={h.hazard} className="border-b border-base-300/80 pb-3 last:border-0">
                <span className="font-medium text-base-content">{display.title}</span>
                {" — "}
                {display.shortDescription}
                {h.hrpRisks && h.hrpRisks.length > 0 && (
                  <p className="mt-1 text-xs text-base-content/75">
                    <span className="font-medium text-base-content/90">
                      {ui.hrpRisksLabel}:{" "}
                    </span>
                    {h.hrpRisks.join(" · ")}
                  </p>
                )}
              </li>
            );
          })}
        </ul>
      </div>
    </div>
  );
}

  } catch {

  }
}

function loadActiveScenarios(): MissionScenarioId[] {
  if (typeof window === "undefined") return [];
  try {
    const raw = localStorage.getItem(SCENARIOS_KEY);
    if (!raw) return [];
    const parsed = JSON.parse(raw) as MissionScenarioId[];
    return Array.isArray(parsed) ? parsed : [];
  } catch {
    return [];
  }
}

function saveActiveScenarios(scenarios: MissionScenarioId[]) {
  try {
    localStorage.setItem(SCENARIOS_KEY, JSON.stringify(scenarios));
  } catch {
    /* ignore */
  }
}

function loadAstronautId(): string {
  if (typeof window === "undefined") return ASTRONAUT_ROSTER[0]?.id ?? "eva-chen";
  try {
    return localStorage.getItem(ASTRONAUT_KEY) ?? ASTRONAUT_ROSTER[0]?.id ?? "eva-chen";
  } catch {
    return ASTRONAUT_ROSTER[0]?.id ?? "eva-chen";
  }
}

function saveAstronautId(id: string) {
  try {
    localStorage.setItem(ASTRONAUT_KEY, id);
  } catch {
    /* ignore */
  }
}

function findLastVisualAttachment(
  logs: VitalReadings[]
): { url: string; missionDay: number } | null {
  for (let i = logs.length - 1; i >= 0; i -= 1) {
    const url = logs[i].imageNoteOrDataUrl;
    if (url?.startsWith("data:image")) {
      return { url, missionDay: logs[i].missionDay };
    }
  }
  return null;
}

function applyScenarioOverlay(
  baseline: VitalReadings[],
  activeScenarios: MissionScenarioId[],
  environment: MissionEnvironment
): VitalReadings[] {
  if (activeScenarios.length === 0 || baseline.length === 0) return baseline;
  const last = baseline[baseline.length - 1];
  const merged = mergeActiveScenarios(last, activeScenarios, environment);
  return [...baseline.slice(0, -1), merged];
}

    return () => {

    };
  }, []);

      const telemetry = feed.telemetry.map((row) => ({ ...row, environment: env }));

      setEnvironment(env);
      saveLogs(telemetry);

        `${feed.crew.callSign} · ${feed.crew.agency} · ${feed.source}`

    );

      /* offline / private mode */
    }

function NumericVitalInput({
  fieldId,
  currentMetric,
  form,
  referenceHint,
  thresholdUnit,
  onChange,
}: {
  fieldId: string;
  currentMetric: VitalsFormMetric;
  form: Partial<VitalReadings>;
  referenceHint: string | null;
  thresholdUnit?: string;
  onChange: (key: string, value: number) => void;
}) {
  const numericValue =
    currentMetric === "radiationMsv"
      ? (form.radiationMsv ?? "")
      : ((form[currentMetric as keyof VitalReadings] as number | undefined) ?? "");

  return (
    <div className="space-y-2">
      <div className="flex items-center gap-2">
        <input
          id={fieldId}
          type="number"
          inputMode="decimal"
          aria-required="true"
          step={
            currentMetric === "temperature" ||
            currentMetric === "sleepHours" ||
            currentMetric === "radiationMsv"
              ? 0.1
              : 1
          }
          className="input input-bordered w-full flex-1"
          value={numericValue}
          onChange={(e) => onChange(currentMetric, Number(e.target.value))}
        />
        {thresholdUnit ? (
          <span className="shrink-0 text-sm font-medium text-base-content/70" aria-hidden>
            {thresholdUnit}
          </span>
        ) : null}
      </div>
      {referenceHint ? (
        <p className="text-xs text-base-content/65">{referenceHint}</p>
      ) : null}
    </div>
  );
}
